import React, { useState, useEffect, useMemo } from 'react';
import { Plus, X, Edit2, Check, Palette, Search } from 'lucide-react';

const COLORS = [
  'bg-yellow-200',
  'bg-blue-200',
  'bg-green-200',
  'bg-pink-200',
  'bg-purple-200',
  'bg-orange-200'
];

const StickyNote = ({ note, onUpdate, onDelete }) => {
  const [isEditing, setIsEditing] = useState(false);
  const [editContent, setEditContent] = useState(note.content);

  const handleSave = () => {
    onUpdate(note.id, editContent);
    setIsEditing(false);
  };

  return (
    <div className={`${note.color} p-4 rounded-md shadow-md w-64 h-64 flex flex-col relative transform transition-transform hover:scale-105`}>
      {/* 刪除與編輯按鈕區 */}
      <div className="absolute top-2 right-2 flex space-x-1">
        {isEditing ? (
          <button onClick={handleSave} className="p-1 text-gray-700 hover:text-green-600 rounded-full hover:bg-white/50 transition-colors">
            <Check size={16} />
          </button>
        ) : (
          <button onClick={() => setIsEditing(true)} className="p-1 text-gray-700 hover:text-blue-600 rounded-full hover:bg-white/50 transition-colors">
            <Edit2 size={16} />
          </button>
        )}
        <button onClick={() => onDelete(note.id)} className="p-1 text-gray-700 hover:text-red-600 rounded-full hover:bg-white/50 transition-colors">
          <X size={16} />
        </button>
      </div>

      {/* 內容區 */}
      <div className="mt-6 flex-grow overflow-hidden">
        {isEditing ? (
          <textarea
            className="w-full h-full bg-transparent resize-none focus:outline-none text-gray-800"
            value={editContent}
            onChange={(e) => setEditContent(e.target.value)}
            autoFocus
            placeholder="在這裡輸入內容..."
          />
        ) : (
          <div className="whitespace-pre-wrap overflow-y-auto h-full text-gray-800 break-words font-medium">
            {note.content || <span className="text-gray-500 italic">空便利貼...</span>}
          </div>
        )}
      </div>
      
      {/* 建立時間顯示 */}
      <div className="text-xs text-gray-500 mt-2 text-right">
        {new Date(note.createdAt).toLocaleDateString()}
      </div>
    </div>
  );
};

export default function StickyNotesApp() {
  const [notes, setNotes] = useState([]);
  const [newNoteContent, setNewNoteContent] = useState('');
  const [selectedColor, setSelectedColor] = useState(COLORS[0]);
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [searchColor, setSearchColor] = useState('');

  useEffect(() => {
    const savedNotes = localStorage.getItem('stickyNotes');
    if (savedNotes) {
      try {
        setNotes(JSON.parse(savedNotes));
      } catch (e) {
        console.error("Failed to parse notes from local storage", e);
      }
    }
  }, []);

  useEffect(() => {
    localStorage.setItem('stickyNotes', JSON.stringify(notes));
  }, [notes]);

  const handleAddNote = (e) => {
    e.preventDefault();
    if (!newNoteContent.trim()) return;

    const newNote = {
      id: Date.now().toString(),
      content: newNoteContent,
      color: selectedColor,
      createdAt: new Date().toISOString()
    };

    setNotes([...notes, newNote]);
    setNewNoteContent('');
    setIsFormOpen(false);
  };

  const handleUpdateNote = (id, newContent) => {
    setNotes(notes.map(note => 
      note.id === id ? { ...note, content: newContent } : note
    ));
  };

  const handleDeleteNote = (id) => {
    if(window.confirm('確定要刪除這張便利貼嗎？')) {
      setNotes(notes.filter(note => note.id !== id));
    }
  };

  const filteredNotes = useMemo(() => {
    return notes.filter(note => {
      const matchQuery = !searchQuery.trim() || note.content.toLowerCase().includes(searchQuery.toLowerCase());
      const matchColor = !searchColor || note.color === searchColor;
      return matchQuery && matchColor;
    });
  }, [notes, searchQuery, searchColor]);

  return (
    <div className="min-h-screen bg-gray-50 font-sans text-gray-900">
      {/* 頂部導航列 */}
      <header className="bg-white shadow-sm sticky top-0 z-10">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-4 flex flex-col lg:flex-row justify-between items-center gap-4">
          <h1 className="text-2xl font-bold text-gray-800 flex items-center gap-2 whitespace-nowrap">
            <Palette className="text-blue-500" />
            便利貼看板
          </h1>
          
          <div className="flex-grow flex flex-col sm:flex-row justify-center items-center gap-3 w-full">
            <div className="relative w-full max-w-md">
              <div className="absolute inset-y-0 left-0 pl-3 flex items-center pointer-events-none">
                <Search className="h-5 w-5 text-gray-400" />
              </div>
              <input
                type="text"
                placeholder="搜尋便利貼內容..."
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                className="block w-full pl-10 pr-3 py-2 border border-gray-300 rounded-md leading-5 bg-white placeholder-gray-500 focus:outline-none focus:placeholder-gray-400 focus:ring-1 focus:ring-blue-500 focus:border-blue-500 sm:text-sm"
              />
              {searchQuery && (
                 <button 
                  onClick={() => setSearchQuery('')}
                  className="absolute inset-y-0 right-0 pr-3 flex items-center text-gray-400 hover:text-gray-600"
                 >
                   <X size={16} />
                 </button>
              )}
            </div>
            
            {/* 顏色篩選器 */}
            <div className="flex items-center gap-1 bg-gray-50 p-1.5 rounded-md border border-gray-200">
              <button
                onClick={() => setSearchColor('')}
                className={`w-7 h-7 rounded-full border border-gray-300 flex items-center justify-center transition-colors ${searchColor === '' ? 'bg-gray-700 text-white shadow-sm' : 'bg-white hover:bg-gray-100 text-gray-500'}`}
                title="全部顏色"
              >
                <span className="text-[10px] font-bold">All</span>
              </button>
              <div className="w-px h-5 bg-gray-300 mx-1"></div>
              {COLORS.map(color => (
                <button
                  key={color}
                  onClick={() => setSearchColor(searchColor === color ? '' : color)}
                  className={`w-7 h-7 rounded-full ${color} border-2 transition-all ${searchColor === color ? 'border-gray-700 scale-110 shadow-md' : 'border-transparent hover:scale-110 shadow-sm'}`}
                  title="篩選此顏色"
                />
              ))}
            </div>
          </div>

          <button 
            onClick={() => setIsFormOpen(!isFormOpen)}
            className="bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded-md shadow flex items-center gap-2 transition-colors font-medium whitespace-nowrap"
          >
            {isFormOpen ? <X size={20} /> : <Plus size={20} />}
            {isFormOpen ? '取消新增' : '新增便利貼'}
          </button>
        </div>
      </header>

      <main className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8">
        
        {}
        {/* 新增便利貼表單區塊 */}
        {isFormOpen && (
          <div className="mb-8 p-6 bg-white rounded-lg shadow-md border border-gray-100 transition-all duration-300 ease-in-out">
            <h2 className="text-lg font-semibold mb-4 text-gray-700">建立新便利貼</h2>
            <form onSubmit={handleAddNote}>
              <textarea
                className="w-full p-3 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500 focus:border-blue-500 min-h-[100px] mb-4 resize-y"
                placeholder="寫下你的想法..."
                value={newNoteContent}
                onChange={(e) => setNewNoteContent(e.target.value)}
                autoFocus
              />
              
              <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
                <div className="flex items-center gap-2">
                  <span className="text-sm text-gray-600 font-medium">選擇顏色：</span>
                  <div className="flex gap-2">
                    {COLORS.map(color => (
                      <button
                        key={color}
                        type="button"
                        onClick={() => setSelectedColor(color)}
                        className={`w-8 h-8 rounded-full ${color} border-2 ${selectedColor === color ? 'border-gray-800 scale-110 shadow-sm' : 'border-transparent hover:scale-110'} transition-all`}
                        aria-label={`選擇顏色 ${color}`}
                      />
                    ))}
                  </div>
                </div>
                
                <button 
                  type="submit"
                  disabled={!newNoteContent.trim()}
                  className="bg-green-600 hover:bg-green-700 disabled:bg-gray-400 disabled:cursor-not-allowed text-white px-6 py-2 rounded-md shadow transition-colors font-medium"
                >
                  儲存便利貼
                </button>
              </div>
            </form>
          </div>
        )}

        {}
        {/* 便利貼網格展示區塊 */}
        {notes.length === 0 ? (
          <div className="text-center py-20 text-gray-500 bg-white rounded-lg shadow-sm border border-dashed border-gray-300">
            <Palette size={48} className="mx-auto text-gray-300 mb-4" />
            <p className="text-xl font-medium text-gray-600">目前沒有便利貼</p>
            <p className="mt-2">點擊右上角的「新增便利貼」開始記錄您的想法！</p>
          </div>
        ) : filteredNotes.length === 0 ? (
          <div className="text-center py-20 text-gray-500 bg-white rounded-lg shadow-sm border border-dashed border-gray-300">
            <Search size={48} className="mx-auto text-gray-300 mb-4" />
            <p className="text-xl font-medium text-gray-600">找不到相符的便利貼</p>
            <p className="mt-2">請嘗試其他搜尋關鍵字。</p>
          </div>
        ) : (
          <div className="flex flex-wrap gap-6 justify-center sm:justify-start">
            {filteredNotes.map(note => (
              <StickyNote 
                key={note.id} 
                note={note} 
                onUpdate={handleUpdateNote}
                onDelete={handleDeleteNote}
              />
            ))}
          </div>
        )}
      </main>
    </div>
  );
}
